2. Prisma Schema — Complete PostgreSQL Schema
This is the complete Prisma schema. GitHub Copilot must put this in: backend/prisma/schema.prisma

generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

model User {
  id          String   @id @default(cuid())
  email       String   @unique
  name        String
  githubId    String   @unique @map("github_id")
  avatarUrl   String?  @map("avatar_url")
  createdAt   DateTime @default(now()) @map("created_at")
  updatedAt   DateTime @updatedAt @map("updated_at")
  memberships TeamMember[]
  queries     Query[]
  @@map("users")
}

model Team {
  id           String   @id @default(cuid())
  name         String
  slug         String   @unique
  plan         String   @default("free")
  createdAt    DateTime @default(now()) @map("created_at")
  members      TeamMember[]
  connections  GithubConnection[]
  queries      Query[]
  prReviews    PRReview[]
  subscription Subscription?
  @@map("teams")
}

model TeamMember {
  id        String   @id @default(cuid())
  role      String   @default("developer")
  joinedAt  DateTime @default(now()) @map("joined_at")
  userId    String   @map("user_id")
  teamId    String   @map("team_id")
  user      User     @relation(fields: [userId], references: [id], onDelete: Cascade)
  team      Team     @relation(fields: [teamId], references: [id], onDelete: Cascade)
  @@unique([userId, teamId])
  @@map("team_members")
}

model GithubConnection {
  id             String    @id @default(cuid())
  teamId         String    @map("team_id")
  repoUrl        String    @map("repo_url")
  repoName       String    @map("repo_name")
  status         String    @default("pending")
  lastIngestedAt DateTime? @map("last_ingested_at")
  createdAt      DateTime  @default(now()) @map("created_at")
  team           Team      @relation(fields: [teamId], references: [id], onDelete: Cascade)
  ingestionJobs  IngestionJob[]
  @@map("github_connections")
}

model IngestionJob {
  id             String    @id @default(cuid())
  teamId         String    @map("team_id")
  connectionId   String    @map("connection_id")
  status         String    @default("pending")
  itemsProcessed Int       @default(0) @map("items_processed")
  totalItems     Int?      @map("total_items")
  errorMessage   String?   @map("error_message")
  startedAt      DateTime? @map("started_at")
  completedAt    DateTime? @map("completed_at")
  createdAt      DateTime  @default(now()) @map("created_at")
  connection     GithubConnection @relation(fields: [connectionId], references: [id])
  @@map("ingestion_jobs")
}

model Query {
  id             String   @id @default(cuid())
  teamId         String   @map("team_id")
  userId         String   @map("user_id")
  conversationId String   @map("conversation_id")
  question       String
  answer         String
  sourcesJson    Json?    @map("sources_json")
  rating         Int?
  createdAt      DateTime @default(now()) @map("created_at")
  team           Team     @relation(fields: [teamId], references: [id])
  user           User     @relation(fields: [userId], references: [id])
  @@map("queries")
}

model PRReview {
  id            String   @id @default(cuid())
  teamId        String   @map("team_id")
  repoName      String   @map("repo_name")
  prNumber      Int      @map("pr_number")
  prUrl         String   @map("pr_url")
  prTitle       String   @map("pr_title")
  reviewSummary String   @map("review_summary")
  issuesJson    Json     @map("issues_json")
  status        String   @default("completed")
  createdAt     DateTime @default(now()) @map("created_at")
  team          Team     @relation(fields: [teamId], references: [id])
  @@map("pr_reviews")
}

model Subscription {
  id                 String    @id @default(cuid())
  teamId             String    @unique @map("team_id")
  plan               String    @default("free")
  status             String    @default("active")
  currentPeriodEnd   DateTime? @map("current_period_end")
  createdAt          DateTime  @default(now()) @map("created_at")
  team               Team      @relation(fields: [teamId], references: [id])
  @@map("subscriptions")
}

4. ChromaDB — Vector Database for AI Embeddings
4.1 How ChromaDB Is Used
ChromaDB stores the AI embeddings (vectors) generated from GitHub code, PRs, and commits. It enables semantic search — finding code that is SIMILAR IN MEANING to a question, not just keyword matches.

Concept
Explanation
MirrorMind Usage
Collection
Like a table in SQL — a named group of vectors
Each team gets one collection named 'team_{teamId}'
Embedding
A list of 384 numbers that represents the 'meaning' of a text
Each code chunk becomes one embedding
Document
The original text that was converted to an embedding
The actual PR text, commit message, or code snippet
Metadata
Extra info stored alongside each vector
source_type, source_url, author, date, repo_name
Query
Search for vectors most similar to a given vector
Find the 20 most relevant code chunks for a developer's question

4.2 ChromaDB Setup in Python AI Service
Create: ai-service/app/services/vectorStore.py
import chromadb
from chromadb.config import Settings

# Connect to ChromaDB running in Docker
client = chromadb.HttpClient(
    host='localhost',
    port=8001  # Note: mapped to 8001 in docker-compose
)

def get_team_collection(team_id: str):
    # Each team has completely isolated storage
    return client.get_or_create_collection(
        name=f'team_{team_id}',
        metadata={'hnsw:space': 'cosine'}
    )

def add_chunks(team_id: str, chunks: list):
    collection = get_team_collection(team_id)
    collection.add(
        ids=[c['id'] for c in chunks],
        embeddings=[c['embedding'] for c in chunks],
        documents=[c['content'] for c in chunks],
        metadatas=[c['metadata'] for c in chunks]
    )

def search_similar(team_id: str, query_embedding: list, n_results: int = 20):
    collection = get_team_collection(team_id)
    results = collection.query(
        query_embeddings=[query_embedding],
        n_results=n_results,
        include=['documents', 'metadatas', 'distances']
    )
    return results

5.2 Query Caching Strategy
To save Groq API calls (free tier has limits), cache AI answers in Redis:
Before calling Groq: check Redis for cached answer using hash of (question + teamId)
If cache hit: return cached answer immediately (no Groq API call needed)
If cache miss: call Groq, generate answer, store in Redis with 1-hour TTL
Expected savings: 30-40% of Groq API calls are repeated questions

6. Data Isolation — Security Between Teams
CRITICAL SECURITY REQUIREMENT: No team must ever see another team's data. This is enforced at the database level.

Database
Isolation Method
How It Works
PostgreSQL
Row-level team_id filter
Every query MUST include WHERE team_id = ? Prisma middleware enforces this automatically.
ChromaDB
Separate collections per team
Each team's vectors are in 'team_{teamId}' collection. Cross-collection search is not possible.
Redis
Namespaced keys with teamId
All Redis keys include team_id. One team cannot read another team's cached data.

Add this Prisma middleware to enforce team isolation automatically:
// In backend/src/app.ts — add to Prisma client setup
prisma.$use(async (params, next) => {
  // Automatically add team_id filter to all queries
  const tablesWithTeamId = ['Query', 'GithubConnection', 'PRReview']
  if (tablesWithTeamId.includes(params.model) && params.action === 'findMany') {
    params.args.where = { ...params.args.where, teamId: getCurrentTeamId() }
  }
  return next(params)
})
